18.1 <stdlib.h> — General Utilities

18.1.1 Sorting with qsort

int compare_int(const void *a, const void *b) {
    int ia = *(const int *)a;
    int ib = *(const int *)b;
    return (ia > ib) - (ia < ib);   // safe — never overflows
    // NOT: return ia - ib;  — overflows for large values
}

int arr[] = {5, 2, 8, 1, 9};
qsort(arr, 5, sizeof(int), compare_int);
// arr = {1, 2, 5, 8, 9}

Struct comparator:

int by_grade_desc(const void *a, const void *b) {
    double ga = ((const Student*)a)->grade;
    double gb = ((const Student*)b)->grade;
    return (ga < gb) - (ga > gb);
}

srand(time(NULL));          // seed ONCE

18.2 <string.h> — Memory Functions

memcpy(dst, src, n);    // copy n bytes — no overlap
memmove(dst, src, n);   // copy n bytes — overlap-safe
memset(ptr, 0, n);      // fill n bytes with 0
memcmp(a, b, n);        // compare n bytes

⚠️ memset zero-init is portable for integers; use calloc for double.

18.3 <math.h> / <cmath>

sqrt(x)  pow(x,y)  fabs(x)  ceil(x)  floor(x)  round(x)
sin(x)   cos(x)    log(x)   exp(x)   atan2(y,x)

abs() from <stdlib.h> is integer-only — use fabs() for double.

Link: gcc prog.c -o prog -lm

18.4 <time.h> — Timing and Dates

// Timing:
clock_t t0 = clock();
do_work();
printf("%.4f s\n", (double)(clock()-t0)/CLOCKS_PER_SEC);

// Date/time:
time_t now  = time(NULL);
struct tm *lt = localtime(&now);
char buf[64];
strftime(buf, sizeof(buf), "%Y-%m-%d %H:%M:%S", lt);
puts(buf);

Argument must be unsigned char or EOF.

18.6 Safe Type Conversion

atoi — unsafe:

int n = atoi("hello");   // returns 0 silently — no error detection!

strtol — safe:

char *end; errno = 0;
long val = strtol("42abc", &end, 10);
if (end == input)        puts("No digits");
else if (*end != '\0')   printf("Trailing: %s\n", end);
else if (errno == ERANGE) puts("Overflow");
else                     printf("OK: %ld\n", val);

C++ alternatives:

int    n = std::stoi("42");         // throws on failure
double d = std::stod("3.14");
std::string s = std::to_string(42);

18.7 C++ STL Glimpse

18.7.1 <vector> — Resizable Array

std::vector<int> v;
v.push_back(10); v.push_back(20);
for (int x : v) std::cout << x << " ";
v.resize(5, 0);

18.7.2 <algorithm>

std::sort(v.begin(), v.end());
std::sort(v.begin(), v.end(), std::greater<int>());
auto it = std::find(v.begin(), v.end(), 10);
int m   = *std::max_element(v.begin(), v.end());

18.7.3 <string>

std::string s = "Alice";
s += " Smith";
s.length();          // 11
s.find("Smith");     // 6
s.substr(6, 5);      // "Smith"

18.8 Key Terms

Term Meaning
qsort C standard sort with function-pointer comparator
bsearch Binary search on sorted array
memcpy Copy bytes (no overlap)
memmove Copy bytes (overlap-safe)
memset Fill bytes
clock() CPU time used by process
strftime Format date/time as string
atoi Unsafe string→int
strtol Safe string→long with error detection
std::vector C++ resizable array
std::sort O(n log n) sort on any range
std::stoi Safe C++ string→int (throws on failure)
